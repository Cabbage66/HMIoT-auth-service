| `content` | String   | 网页内容摘要         | `page content summary...` |

#### 示例数据

```csv
id,date,user,pc,url,content
{...},01/02/2010 09:00:00,ABC0123,PC-1234,http://news.site.com,news article content...
```

#### 提取特征

- `www_visit_count` - 网页访问次数
- `www_download_count` - 下载次数（从URL模式推断）
- `www_upload_count` - 上传次数（从URL模式推断）
- `unique_domains` - 访问的不同域名数量
- `content_diversity` - 内容多样性（基于访问的网站类型）

# 数据表（补）

`-- 1. 用户登录历史日志`
`CREATE TABLE IF NOT EXISTS user_login_log (`
    `log_id        BIGINT AUTO_INCREMENT PRIMARY KEY,`
    `user_id       BIGINT NOT NULL,`
    `username      VARCHAR(64) NOT NULL,`
    `login_time    DATETIME NOT NULL,`
    `device_id     VARCHAR(128) COMMENT '设备ID/指纹',`
    `login_ip      VARCHAR(64),`
    `login_result  VARCHAR(20) COMMENT 'success/failed',`
    `created_at    DATETIME DEFAULT CURRENT_TIMESTAMP,`
    `INDEX idx_user_time (user_id, login_time),`
    `INDEX idx_username_time (username, login_time)`
`) ENGINE=InnoDB COMMENT='用户登录历史日志';`

`-- 2. 文件操作日志`
`CREATE TABLE IF NOT EXISTS file_operation_log (`
    `log_id         BIGINT AUTO_INCREMENT PRIMARY KEY,`
    `user_id        BIGINT NOT NULL,`
    `username       VARCHAR(64) NOT NULL,`
    `operation_time DATETIME NOT NULL,`
    `device_id      VARCHAR(128),`
    `file_path      VARCHAR(500),`
    `file_name      VARCHAR(255),`
    `operation_type VARCHAR(20) COMMENT 'read/write/delete/download',`
    `created_at     DATETIME DEFAULT CURRENT_TIMESTAMP,`
    `INDEX idx_user_time (user_id, operation_time)`
`) ENGINE=InnoDB COMMENT='文件操作日志';`

`-- 3. 邮件发送日志（可选）`
`CREATE TABLE IF NOT EXISTS email_log (`
    `log_id      BIGINT AUTO_INCREMENT PRIMARY KEY,`
    `sender_id   BIGINT NOT NULL,`
    `sender_name VARCHAR(64),`
    `send_time   DATETIME NOT NULL,`
    `recipient   VARCHAR(255),`
    `subject     VARCHAR(500),`
    `email_size  INT COMMENT '邮件大小（字节）',`
    `created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,`
    `INDEX idx_sender_time (sender_id, send_time)`
`) ENGINE=InnoDB COMMENT='邮件发送日志';`

`-- 4. HTTP 访问日志`
`CREATE TABLE IF NOT EXISTS http_access_log (`
    `log_id      BIGINT AUTO_INCREMENT PRIMARY KEY,`
    `user_id     BIGINT NOT NULL,`
    `username    VARCHAR(64),`
    `access_time DATETIME NOT NULL,`
    `device_id   VARCHAR(128),`
    `url         VARCHAR(500),`
    `method      VARCHAR(10),`
    `created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,`
    `INDEX idx_user_time (user_id, access_time)`
`) ENGINE=InnoDB COMMENT='HTTP访问日志';`

`--  5. 设备操作日志`
`CREATE TABLE IF NOT EXISTS device_operation_log (`
    `log_id         BIGINT AUTO_INCREMENT PRIMARY KEY,`
    `user_id        BIGINT NOT NULL,`
    `username       VARCHAR(64),`
    `operation_time DATETIME NOT NULL,`
    `device_id      VARCHAR(128),`
    `operation_type VARCHAR(20) COMMENT 'connect/disconnect/mount/unmount',`
    `created_at     DATETIME DEFAULT CURRENT_TIMESTAMP,`
    `INDEX idx_user_time (user_id, operation_time)`
`) ENGINE=InnoDB COMMENT='设备操作日志';`
